\begin{defbox}\boxtitle{לפני שמתחילים – איך בכלל מדברים עם נתב? (למי שאין רקע)}\par  כל הפרק הזה עוסק ב"הקשחת נתב" – אבל איך פונים לנתב מלכתחילה? הנה הבסיס:  \textbf{\enLR{CLI (Command Line Interface)}} – ממשק שורת פקודה. במקום עכבר וחלונות, מקלידים פקודות טקסט. כל ציוד סיסקו מנוהל כך. \textbf{\enLR{IOS}} – מערכת ההפעלה של ציוד סיסקו (כמו \enLR{Windows} למחשב, רק לנתב). ה-\enLR{CLI} הוא הדרך לדבר איתה. \textbf{מצבי עבודה \enLR{(Modes)}} – ל-\enLR{CLI} יש "רמות". הסימן שאחרי שם הנתב מראה איפה אתם:  \code{R1>} – מצב בסיסי \enLR{(User)} – רק צפייה. \code{R1\#} – מצב מנהל \enLR{(Privileged)} – אחרי הקלדת \code{enable}. אפשר הכול. \code{R1(config)\#} – מצב הגדרות – אחרי \code{configure terminal}. כאן משנים הגדרות. \code{R1(config-if)\#} – מצב ממשק – מגדירים כרטיס רשת ספציפי.  \textbf{קונפיגורציה \enLR{(Configuration)}} – כל ההגדרות של הנתב. יש \textbf{\enLR{running-config}} (הפעילה כרגע בזיכרון, נמחקת בכיבוי) ו-\textbf{\enLR{startup-config}} (השמורה, נטענת באתחול). שומרים עם \code{copy running-config startup-config} או \code{write}. \textbf{ממשק \enLR{(Interface)}} – כרטיס רשת/פורט פיזי בנתב, למשל \code{GigabitEthernet0/0} (בקיצור \enLR{g0/0).}  \textbf{דוגמה – כניסה והגדרה ראשונה:} \enLR{Router> enable} ! מעבר למצב מנהל \enLR{Router\# configure terminal} ! מעבר למצב הגדרות \enLR{Router(config)\# hostname R1} ! שינוי שם הנתב \enLR{R1(config)\# exit R1\# write} ! שמירת ההגדרות\end{defbox}

\sectionheading{\enLR{2.1} למה הנתב הוא המטרה הראשונה}

\textbf{נתב הקצה} \enLR{(Edge router)} הוא הנקודה שבה רשת הארגון פוגשת את האינטרנט. מי ששולט בו יכול לנתב תעבורה לאן שירצה, להאזין לה, לחסום אותה או לפתוח פתח פנימה. שלוש גישות לתכנון:\par

\begin{itemize}
\item \textbf{נתב יחיד} – הנתב הוא גם חומת האש \enLR{(ACL, ZPF).} מתאים לעסק קטן \enLR{(SOHO).}
\item \textbf{הגנה לעומק \enLR{(Defense in depth)}} – נתב קצה מסנן גס ⇒ חומת אש ייעודית \enLR{(ASA)} ⇒ נתב פנימי. כל שכבה עצמאית.
\item \textbf{\enLR{DMZ}} – אזור מפורז בין שתי חומות אש (או בין שני ממשקים של אותה חומה) ובו השרתים הציבוריים \enLR{(web, mail).} אם שרת ב-\enLR{DMZ} נפרץ, התוקף עדיין לא בפנים.
\end{itemize}

אבטחת הנתב עצמו כוללת שלושה תחומים: \textbf{אבטחה פיזית} (חדר נעול, \enLR{UPS} – מי שיש לו גישה פיזית לקונסול יכול לאפס סיסמה דרך \enLR{password recovery)}, \textbf{אבטחת מערכת ההפעלה} (גרסת \enLR{IOS} עדכנית, זיכרון מספיק, גיבוי של ה-\enLR{image} והקונפיגורציה) ו\textbf{הקשחה} \enLR{(hardening)} – סגירת שירותים מיותרים, סיסמאות, גישה מוצפנת, ניטור. זה ליבת הפרק.\par

\begin{storybox}\boxtitle{סיפור מהחיים: הנתבים שנשארו עם \enLR{admin/admin}}\par  ב-\enLR{2016} רשת \enLR{Mirai} הדביקה \enLR{600,000} מכשירים – מצלמות אבטחה, ראוטרים ביתיים, מקליטי \enLR{DVR} – בשיטה פרימיטיבית להדהים: היא ניסתה \enLR{62} צירופים של שם משתמש וסיסמה מברירת המחדל של היצרנים \enLR{(admin/admin, root/12345, admin/password).} זהו. בלי פגיעות מתוחכמת. עם הצבא הזה הופלו ב-\enLR{DDoS} של \enLR{1.2} טרה-ביט/שנייה שירות ה-\enLR{DNS} של \enLR{Dyn}, ואיתו \enLR{Twitter, Netflix, Spotify} ו-\enLR{PayPal} ליום שלם. כל מה שנלמד בפרק הזה – סיסמה חזקה, \enLR{SSH} במקום \enLR{Telnet}, כיבוי שירותים – היה מונע את ההדבקה. שאלו: כמה מכם שינו את הסיסמה של הראוטר בבית?\end{storybox}

\begin{faqbox}\boxtitle{שאלת תלמיד: "אם הגדרתי גם \enLR{enable password} וגם \enLR{enable secret} – מה קובע?"}\par  ה-\enLR{secret.} ה-\enLR{IOS} מתעלם מ-\enLR{enable password} כשקיים \enLR{enable secret} (ומזהיר אם הן זהות). למען הסדר – פשוט לא מגדירים \enLR{enable password.}\end{faqbox}

\begin{faqbox}\boxtitle{שאלת תלמיד: "מה ההבדל בין \enLR{login} ל-\enLR{login local}?"}\par  \code{login} – מבקש רק את הסיסמה שהוגדרה בקו עצמו (\code{password} בתוך \enLR{line).} כולם משתמשים באותה סיסמה, אין שמות, אין מעקב מי נכנס.\newline  \code{login local} – מבקש שם משתמש וסיסמה ומאמת מול מסד המשתמשים המקומי (\code{username … secret …}). \textbf{\enLR{SSH} דורש שם משתמש ולכן חייב \enLR{login local} (או \enLR{AAA).}} זה בדיוק מה שנשאל בבגרות שאלה \enLR{2}א: התשובה הנכונה היא \code{username admin privilege 15 secret …} + \code{login local}.\end{faqbox}

\sectionheading{\enLR{2.4 Telnet} מול \enLR{SSH} – והגדרת \enLR{SSH} צעד אחר צעד}

\textbf{\enLR{Telnet}} \enLR{(TCP 23)} מעביר הכול – כולל שם משתמש וסיסמה – בטקסט גלוי. כל מי שמאזין לרשת \enLR{(Wireshark)} רואה אותם. \textbf{\enLR{SSH}} \enLR{(Secure Shell, TCP 22)} מצפין את כל ההפעלה ומאמת את השרת. \enLR{SSH} גרסה \enLR{2} בלבד (גרסה \enLR{1} פגיעה).\par

כדי ש-\enLR{SSH} יעבוד, הנתב צריך \textbf{זוג מפתחות \enLR{RSA}} (פרק \enLR{7} יסביר לעומק: מפתח ציבורי שנשלח ללקוח ומפתח פרטי שנשאר בנתב). כדי לייצר מפתח הנתב צריך שם מלא \enLR{(FQDN) = hostname + domain-name}, כי השם נכנס למפתח.\par

R1(config-line)\# \textcolor{cbold}{\textbf{login local}}\\

PC> ssh -l admin 10.0.0.1
\end{english}\end{codebox}

\begin{mistakebox}\boxtitle{טעויות נפוצות בהגדרת \enLR{SSH}}\par  • שכחו \code{ip domain-name} ⇒ הפקודה \code{crypto key generate rsa} נכשלת ("\enLR{Please define a domain-name first").}\newline  • השאירו \enLR{hostname "Router"} ⇒ אותה שגיאה.\newline  • הגדירו \code{login} במקום \code{login local} ⇒ \enLR{SSH} דוחה ("\enLR{Password required, but none set"} / אין אימות בשם משתמש).\newline  • מפתח קטן מ-\enLR{768} ביט לא מאפשר \enLR{SSHv2.} משתמשים ב-\enLR{1024} לפחות; מומלץ \enLR{2048.}\newline  • \code{transport input ssh} חוסם \enLR{Telnet} – אם התלמיד היה מחובר ב-\enLR{Telnet} הוא לא יינתק, אבל חיבור הבא ייכשל. זה מכוון.\end{mistakebox}

\begin{exambox}\boxtitle{בבגרות (שאלה \enLR{5}ב)}\par  "מנהל רשת הפיק מפתחות \enLR{RSA} בנתב. מה ניתן לעשות כעת עם המפתחות?" – \textbf{ניתן להשתמש בהם לצורך חיבור \enLR{SSH}} (ללא צורך באיפוס). \enLR{Telnet} לא משתמש במפתחות כלל.\end{exambox}

\begin{storybox}\boxtitle{סיפור מהחיים: \enLR{Telnet} בבית חולים}\par  בבדיקת חדירה בבית חולים (סיפור שמופיע בדוחות של חברות אבטחה רבות), הבודקים חיברו מחשב נייד לשקע רשת בחדר המתנה, הריצו \enLR{Wireshark}, וחיכו. תוך \enLR{40} דקות הם לכדו \enLR{session} של \enLR{Telnet} של טכנאי שהתחבר למתג הראשי – שם משתמש וסיסמה בטקסט גלוי. עם הסיסמה הזו הם קיבלו שליטה בכל המתגים בבניין. הטכנאי לא עשה שום דבר "לא נכון" – הוא פשוט השתמש בכלי שהיה זמין. ההגנה: \code{transport input ssh} – שורה אחת.\end{storybox}

\sectionheading{\enLR{2.9 SNMP} – ניטור ציוד, ולמה גרסה \enLR{3}}

\textbf{\enLR{SNMP}} \enLR{(Simple Network Management Protocol, UDP 161/162)} מאפשר לתחנת ניהול \enLR{(NMS} – \enLR{PRTG, SolarWinds, Zabbix)} לקרוא נתונים מהנתב \enLR{(GET} – תעבורה, \enLR{CPU)} ואף לשנות \enLR{(SET).} הנתב יכול גם לשלוח התראות יזומות (\textbf{\enLR{trap}}). הנתונים מאורגנים ב-\textbf{\enLR{MIB}} (עץ של משתנים, כל אחד עם מזהה \enLR{OID).}\par

\begin{xltabular}{\linewidth}{|>{\setRTL\raggedleft\arraybackslash}X|>{\setRTL\raggedleft\arraybackslash}X|>{\setRTL\raggedleft\arraybackslash}X|>{\setRTL\raggedleft\arraybackslash}X|}

\textbf{היחיד המומלץ}. רמות: \enLR{noAuthNoPriv, authNoPriv, authPriv} & \enLR{DES/AES} & שם משתמש + \enLR{HMAC (MD5/SHA)} & \textbf{\enLR{SNMPv3}} \\

\sectionheading{\enLR{2.13 SDM / CCP} – מה זה ומה אומרים לתלמידים}

תכנית הלימודים מזכירה "\enLR{SDM} לוריפיקציה". \textbf{\enLR{SDM}} \enLR{(Security Device Manager)} היה ממשק גרפי מבוסס \enLR{Java} שרץ בדפדפן מול הנתב והציע אשפים \enLR{(wizards)} ל-\enLR{SSH}, חומת אש, \enLR{VPN} ו-"\enLR{Security Audit"} – סריקה שמציגה רשימת שירותים מסוכנים ומציעה לתקן בלחיצה ("\enLR{One-step lockdown"}, המקבילה הגרפית של \enLR{AutoSecure). \sectionheading{בנק שאלות שתלמידים שואלים – ותשובות מוכנות}

שאלות נפוצות בפרק הקשחת הנתב, עם תשובות מוכנות.\par

\subheading{סיסמאות וגישה}

\textbf{ש: "אם אני שם סיסמה חזקה ל-\enLR{Telnet}, למה בכל זאת צריך \enLR{SSH}?"}\newline ת: כי \enLR{Telnet} מעביר את \underline{הכול} בטקסט גלוי, כולל את הסיסמה החזקה עצמה. כל מי שמאזין לרשת \enLR{(Wireshark)} רואה אותה. \enLR{SSH} מצפין את כל התקשורת. חוזק הסיסמה לא עוזר אם היא נשלחת גלויה.\par

\textbf{ש: "מה ההבדל בין \code{password} ל-\code{secret}?"}\newline ת: \code{secret} נשמר כ\underline{גיבוב חד-כיווני} (לא ניתן לשחזר). \code{password} נשמר גלוי, ואפילו עם \code{service password-encryption} הוא רק "\enLR{type 7"} שמפוענח באתר אונליין בשנייה. תמיד להשתמש ב-\code{secret}.\par

\textbf{ש: "אם \code{service password-encryption} זו הצפנה, למה היא חלשה?"}\newline ת: כי זה אלגוריתם הפיך וישן (וריאציה על \enLR{Vigen}è\enLR{re).} הוא מגן רק מפני מישהו שמציץ מעבר לכתף בזמן שאתם צופים בקונפיג – לא מפני תוקף אמיתי. ה"חוזק" האמיתי הוא ב-\code{secret} (גיבוב).\par

\textbf{ש: "מה קורה אם שכחתי את ה-\enLR{enable secret}?"}\newline ת: יש הליך \textbf{\enLR{password recovery}} דרך גישה פיזית לקונסול: מאתחלים, נכנסים ל-\enLR{ROMMON}, משנים את ה-\enLR{configuration register (0x2142)} כדי לדלג על ה-\enLR{startup-config}, ומגדירים סיסמה חדשה. המסקנה החשובה: \underline{מי שיש לו גישה פיזית לנתב – שולט בו}, ולכן אבטחה פיזית קריטית.\par

\textbf{ש: "מה ההבדל בין רמות הרשאה \enLR{(privilege levels)} לתצוגות \enLR{(views)}?"}\newline ת: רמות הרשאה גסות – רמה נתונה מקבלת אוטומטית גם את כל מה שמתחתיה. תצוגות \enLR{(parser view)} מדויקות – מגדירים רשימת פקודות מדויקת לכל תפקיד, אך דורשות שה-\enLR{AAA} יופעל.\par

\subheading{\enLR{SSH}}

\textbf{ש: "למה צריך \code{ip domain-name} כדי ליצור מפתחות \enLR{SSH}?"}\newline ת: כי שם המפתח נגזר מהשם המלא של הנתב = \enLR{hostname + domain-name (FQDN).} בלי דומיין הנתב לא יודע איזה שם לתת למפתח, והפקודה נכשלת עם "\enLR{Please define a domain-name first".}\par

\textbf{ש: "אחרי \code{transport input ssh} ניתקתי בטעות את \enLR{Telnet} – זו טעות?"}\newline ת: לא, זה בדיוק מה שרוצים – לחסום את \enLR{Telnet} הלא-מאובטח ולאפשר רק \enLR{SSH.} חיבור \enLR{Telnet} חדש ייכשל, וזו התנהגות רצויה.\par

\subheading{שירותים וניטור}

\textbf{ש: "למה זמן מדויק \enLR{(NTP)} קשור לאבטחה בכלל?"}\newline ת: כי בלי זמן מסונכרן, הלוגים מכל המכשירים לא ניתנים לשחזור – אי אפשר לדעת מה קרה לפני מה כשחוקרים אירוע. בנוסף, תעודות דיגיטליות ומפתחות \enLR{VPN} תלויים בזמן נכון.\par

\textbf{ש: "\enLR{SNMPv2} באמת כל כך גרוע?"}\newline ת: הבעיה שלו היא ה-\enLR{community string} שעובר בטקסט גלוי (ולעיתים נשאר "\enLR{public").} אם חייבים \enLR{v2c} – לפחות \enLR{read-only} + הגבלה ב-\enLR{ACL} לכתובת תחנת הניהול. אבל הנכון הוא \enLR{v3}, שמוסיף אימות \enLR{(HMAC)} והצפנה.\par

\textbf{ש: "אז למה שלא נריץ תמיד \code{auto secure} ונגמור עם זה?"}\newline ת: כי הוא "עיוור" – יכול לכבות \enLR{CDP} שהמתגים צריכים, להפעיל חומת אש שתחסום תעבורה לגיטימית, ואי אפשר לבטלו בפקודה אחת. מריצים אותו, בודקים מה שינה עם \code{show auto secure config}, ומכווננים.\par

\textbf{ש: "מה זה בעצם \enLR{CLI}, ולמה לא ללמוד ממשק גרפי?"}\newline ת: \enLR{CLI} הוא ממשק שורת הפקודה של הנתב. אנחנו לומדים אותו כי הוא אוניברסלי (עובד על כל ציוד סיסקו וגם דרך \enLR{SSH)}, מדויק, וניתן לאוטומציה – \underline{והבגרות בודקת אותו בלבד}. הממשקים הגרפיים \enLR{(SDM/CCP)} יצאו משימוש.\par
